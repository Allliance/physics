\subsubsection*{Solution}

{\color{blue}
The limit must be understood as $\omega\to0^+$ while retaining the leading Fermi-liquid term proportional to $\omega^2$. At exactly $\omega=0$, both inelastic rates vanish.

Let
\[
x\equiv ak_F\ll1.
\]
Near the bottom of the four-dimensional nearest-neighbor band,
\[
\varepsilon_{\mathbf k}-\varepsilon_{\min}
 =ta^2k^2-\frac{ta^4}{12}\sum_{\alpha=1}^{4}k_\alpha^4+O(a^6k^6),
\qquad
\varepsilon_F\sim t x^2.
\]
The density, including the two spin species, is
\[
n=2\frac{(\pi^2/2)k_F^4}{(2\pi)^4}
 =\frac{k_F^4}{16\pi^2},
\]
so $n\propto k_F^4$.

\paragraph*{Quasiparticle rate}
The second-order dilute-limit scaling of the on-shell decay rate in $d$ spatial dimensions is
\[
\Gamma_{\mathrm{qp}}(k_F,\omega)
 \sim \frac{U^2}{t}\,x^{2d-2}
       \left(\frac{\omega}{\varepsilon_F}\right)^2.
\]
For $d=4$ and $\varepsilon_F\sim tx^2$,
\[
\Gamma_{\mathrm{qp}}(k_F,\omega)
 \sim \frac{U^2}{t^3}\,x^2\omega^2,
\]
and hence
\[
A_{\mathrm{qp}}(k_F)
 =\lim_{\omega\to0^+}\frac{\Gamma_{\mathrm{qp}}}{\omega^2}
 \propto k_F^2.
\]
Thus $\gamma_{\mathrm{qp}}=2$.

\paragraph*{Transport rate}
The band velocity is
\[
v_y(\mathbf k)=2ta\sin(ak_y)
 =2ta^2k_y-\frac{ta^4}{3}k_y^3+O(a^6k_y^5).
\]
For a two-particle collision, momentum conservation cancels the contribution from the linear-in-$k_y$ velocity. The first current-relaxing term comes from the cubic term, so
\[
\frac{\Delta v_y}{v_F}\sim (ak_F)^2=x^2.
\]
A transport rate contains the square of this fractional current change. Therefore
\[
\Gamma_{\mathrm{tr}}\sim x^4\Gamma_{\mathrm{qp}}
 \sim \frac{U^2}{t^3}\,x^6\omega^2,
\]
which gives
\[
A_{\mathrm{tr}}(k_F)\propto k_F^6.
\]
Thus $\gamma_{\mathrm{tr}}=6$.

\paragraph*{Regular optical conductivity}
Writing the low-frequency conductivity in extended-Drude form, its regular part at order $U^2$ is
\[
\delta\operatorname{Re}\sigma^{\mathrm{reg}}_{yy}(0^+)
 =K_y A_{\mathrm{tr}}+O(U^4),
\]
where the optical weight per unit physical product of the two two-dimensional momentum-space measures obeys
\[
K_y\sim\frac{ne^2}{m}\propto k_F^4.
\]
Consequently,
\[
\delta\operatorname{Re}\sigma^{\mathrm{reg}}_{yy}(0^+)
 \propto k_F^4k_F^6=k_F^{10},
\]
so $\gamma_\sigma=10$.

Equivalently, up to a positive dimensionless phase-space constant,
\[
\delta\operatorname{Re}\sigma^{\mathrm{reg}}_{yy}(0^+)
 \sim \frac{e^2}{\hbar a^2}\left(\frac{U}{t}\right)^2(ak_F)^{10}.
\]

The definition of the rates is essential: at fixed reduced frequency $\omega/\varepsilon_F$, the rates themselves scale instead as $k_F^6$ and $k_F^{10}$. The requested exponents $2$ and $6$ are those of the coefficients of $\omega^2$ in the sequential limit in which the low-frequency coefficient is extracted at fixed \(k_F\), before its \(k_F\to0\) scaling is taken.
