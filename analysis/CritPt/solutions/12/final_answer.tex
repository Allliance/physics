\section*{Final answer}


\[
\boxed{
|\psi^f(q)\rangle
=
\exp\!\left[
\frac{i\pi}{N}
(q-k_{12})
\left(
q+k_{12}+N+2k_{23}-2k_{13}
\right)
\right]
|\psi^i(q)\rangle
}
\]

or, equivalently,

\[
\boxed{
|\psi^f(q)\rangle
=
\exp\!\left\{
\frac{i\pi}{N}
\left[
\left(q+k_{23}-k_{13}+\frac N2\right)^2
-
\left(k_{12}+k_{23}-k_{13}+\frac N2\right)^2
\right]
\right\}
|\psi^i(q)\rangle .
}
\]
